\subsection{Proof of Theorem~\ref{thm:finite-sample}}

\begin{proof}
For fixed \((d,b)\), the variables
\(U_1^d(b),\ldots,U_n^d(b)\) are independent across task instances, lie in
\([0,1]\), and have mean \(F_d(b)\).  This remains true when retries and paired
design runs are dependent within an instance.  Hoeffding's inequality gives
\[
\Prob\!\left(\left|\widehat F_d(b)-F_d(b)\right|>u\right)
\le 2e^{-2nu^2}.
\]
At \(u=\eta_n(\alpha)\), the right-hand side is
\(\alpha/(|\mathcal D||\mathcal B|)\).  A union bound over all designs and
declared grid budgets proves Equation~\eqref{eq:simultaneous-bands}.

On this event, for every \(b<T_d^-\), \(U_d(b)<1-\delta\), so
\(F_d(b)<1-\delta\) and \(T_d^-\le T_d(\delta)\).  At \(b=T_d^+\),
\(L_d(b)\ge1-\delta\), so \(F_d(b)\ge1-\delta\) and
\(T_d(\delta)\le T_d^+\), up to grid resolution.  The admissibility statement
follows from \(F_d(B)\ge L_d(B)\).  Finally, if
\(T_{\widehat d}^+<T_d^-\) for every competitor, then
\(T_{\widehat d}(\delta)\le T_{\widehat d}^+<T_d^-\le T_d(\delta)\), proving
that \(\widehat d\) is the unique winner.
\end{proof}

\subsection{Proof of Proposition~\ref{prop:retry-first-passage}}

\begin{proof}
Conditional on \((x,z)\), retry success indicators are IID Bernoulli with
parameter \(\lambda_d(x,z)>0\), hence \(R_d(x,z)\) is geometric on
\(\{1,2,\ldots\}\).  The first identity is the resource ledger: the design
pays any incremental signal cost and every realized retry cost through the
first verified artifact.  Moreover,
\begin{align*}
\E\!\left[\sum_{r=1}^{R_d}K_{d,r}\,\middle|\,x,z\right]
&=\sum_{r\ge1}\E[K_{d,r}\mathbf 1\{R_d\ge r\}\mid x,z]\\
&=\overline K_d(x,z)\sum_{r\ge1}\{1-\lambda_d(x,z)\}^{r-1}\\
&=\frac{\overline K_d(x,z)}{\lambda_d(x,z)}.
\end{align*}
The second equality holds because \(\{R_d\ge r\}\) depends only on the first
\(r-1\) retry outcomes and is therefore independent of current cost
\(K_{d,r}\) under the IID-pair assumption.  Adding \(c_d(z)\) proves
Equation~\eqref{eq:variable-cost-first-passage}.
\end{proof}

\subsection{Proof of Proposition~\ref{prop:tool-first-passage}}

\begin{proof}
In the reference-sequence regime, a complete retry succeeds only when the tool
policy reproduces \(\mathbf a^\star(x)\), an event of probability
\(q_d^\star(x,z)\), and the resulting execution is verified with conditional
probability \(p_d(x)\).  Therefore
\(\lambda_d(x,z)=q_d^\star(x,z)p_d(x)\).  Substituting this identity and
\(t_0(d,x)=1/p_d(x)\) into Proposition~\ref{prop:retry-first-passage}, with no
separate signal cost, proves Equation~\eqref{eq:tool-first-passage}.
\end{proof}

\subsection{Proof of Lemma~\ref{lem:tool-cross-entropy}}

\begin{proof}
At decision \(k\), conditional on \(L\ge k\), the cross-entropy identity gives
\begin{align*}
&\E[-\log q_d(A_k^\star\mid Z,H_k^\star)\mid L\ge k]\\
&\quad=H(A_k^\star\mid Z,H_k^\star,L\ge k)
+\E\!\left[\KL\!\left(\rho_k(\cdot\mid Z,H_k^\star)
\Vert q_d(\cdot\mid Z,H_k^\star)\right)\middle|L\ge k\right].
\end{align*}
The definition of conditional mutual information also gives
\[
H(A_k^\star\mid Z,H_k^\star,L\ge k)
=H(A_k^\star\mid H_k^\star,L\ge k)
-I(A_k^\star;Z\mid H_k^\star,L\ge k).
\]
Multiplying by \(\Prob(L\ge k)\) and summing over \(k\) proves
Equation~\eqref{eq:tool-cross-entropy}.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:four-term}}

\begin{proof}
Taking logarithms in Equation~\eqref{eq:tool-first-passage}, then averaging over
\((X,Z)\), gives
\begin{align*}
\mathcal L(d)
={}&\E_{X,Z}\log\overline K_d(X,Z)+\E_X\log t_0(d,X)\\
&+\E\!\left[-\sum_{k=1}^{L}
\log q_d(A_k^\star\mid Z,H_k^\star)\right].
\end{align*}
Lemma~\ref{lem:tool-cross-entropy} turns the last term into
\(\mathcal H_{\mathrm{ref}}-G_Z+\varepsilon_d\).  Applying the same expression
to \(d_0\) and subtracting cancels the common reference entropy and yields
Equation~\eqref{eq:four-term-general}.  If \(Z_0\) is conditionally
uninformative, \(G_{Z_0}=0\); if its policy matches the corresponding
reference-tool prior, \(\varepsilon_{d_0}=0\).  This proves
Equation~\eqref{eq:four-term-baseline}.
\end{proof}

\subsection{Proof of Proposition~\ref{prop:off-reference}}

\begin{proof}
Reference and non-reference success are disjoint, so the actual complete-retry
success probability is
\[
q_d^\star(x,z)p_d(x)+r_d^{\mathrm{alt}}(x,z).
\]
Excluding separate signal cost, Proposition~\ref{prop:retry-first-passage}
therefore gives actual conditional mean
\(\overline K_d(x,z)/\{q_d^\star(x,z)p_d(x)+r_d^{\mathrm{alt}}(x,z)\}\),
whereas the reference-sequence regime predicts
\(\overline K_d(x,z)/\{q_d^\star(x,z)p_d(x)\}\).  The logarithm of the first
quantity minus the logarithm of the second is exactly
Equation~\eqref{eq:off-reference-residual}.  If the displayed hazard ratio is
at most \(\xi\), monotonicity of \(\log(1+u)\) gives the stated bound.
\end{proof}
