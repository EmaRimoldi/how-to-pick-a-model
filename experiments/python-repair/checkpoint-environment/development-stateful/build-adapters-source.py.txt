def extract(text,names):
    lines=text.splitlines(True);pieces=[];segments=[]
    for node in ast.parse(text).body:
        name=getattr(node,'name',None)
        if isinstance(node,ast.Assign) and len(node.targets)==1:name=getattr(node.targets[0],'id',None)
        if name in names:
            piece=''.join(lines[node.lineno-1:node.end_lineno]);pieces.append(piece)
            segments.append({'start_line':node.lineno,'end_line':node.end_lineno,'sha256':hashlib.sha256(piece.encode()).hexdigest()})
    assert len(pieces)==len(names)
    return '\n\n'.join(pieces),segments

def sources(slug,folder,names):
    raw=json.loads((folder/'sources.json').read_text());src={}
    sub=ROOT/'upstream'/slug;sub.mkdir(parents=True)
    cache=CACHE/'scrapy'/str(int(slug.split('-')[1]));cache.mkdir(parents=True)
    for role,meta in raw['files'].items():
        path=folder/(role+'.txt');data=path.read_bytes();assert hashlib.sha256(data).hexdigest()==meta['sha256']
        dest=sub/(role+'-full.txt');dest.write_bytes(data);(cache/(role+'.txt')).write_bytes(data)
        src[role]={**meta,'local_path':str(dest.relative_to(ROOT))}
        if role in ('buggy','fixed'):
            text,segments=extract(data.decode(),names);ex=sub/(role+'.txt');ex.write_text(text)
            src[role].update(excerpt=str(ex.relative_to(ROOT)),excerpt_sha256=hashlib.sha256(text.encode()).hexdigest(),segments=segments)
    return src

def piece(folder,role,names):return extract((folder/(role+'.txt')).read_text(),names)[0]
